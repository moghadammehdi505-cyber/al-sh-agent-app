import streamlit as st
from google import genai

# ۱. تنظیمات اولیه صفحه وب (عنوان و آیکون مرورگر)
st.set_page_config(
    page_title="داشبورد ایجنت هوش مصنوعی", 
    page_icon="🤖", 
    layout="centered"
)

st.title("🤖 دستیار هوشمند و ایجنت من")
st.write("این داشبورد آنلاین به مدل جمنای گوگل متصل است.")

# ۲. ایجاد منوی کناری (Sidebar) برای دریافت کلید API از کاربر
with st.sidebar:
    st.header("تنظیمات اتصال")
    api_key = st.text_input("کلید API گوگل (Gemini API Key):", type="password")
    st.markdown("---")
    st.info("کلید API خود را از Google AI Studio دریافت کنید و اینجا وارد نمایید.")

# ۳. بررسی اینکه آیا کاربر کلید API را وارد کرده است یا خیر
if not api_key:
    st.warning("⚠️ لطفاً برای شروع کار، کلید API خود را در منوی کناری وارد کنید.")
else:
    # راه‌اندازی کلاینت گوگل با کلید وارد شده
    client = genai.Client(api_key=api_key)

    # ۴. مدیریت حافظه و تاریخچه پیام‌ها در صفحه
    if "messages" not in st.session_state:
        st.session_state.messages = []

    # نمایش پیام‌های قبلی در صفحه چت
    for message in st.session_state.messages:
        with st.chat_message(message["role"]):
            st.markdown(message["content"])

    # ۵. دریافت پیام جدید از کاربر در کادر پایین صفحه
    if prompt := st.chat_input("سوال یا دستور خود را بنویسید..."):
        
        # ذخیره و نمایش پیام کاربر
        st.session_state.messages.append({"role": "user", "content": prompt})
        with st.chat_message("user"):
            st.markdown(prompt)

        # ۶. ارسال درخواست به مدل جمنای گوگل و دریافت پاسخ
        with st.chat_message("assistant"):
            with st.spinner("ایجنت در حال تفکر و پردازش است..."):
                try:
                    # فراخوانی مدل سریع و استاندارد جمنای
                    response = client.models.generate_content(
                        model="gemini-2.5-flash",
                        contents=prompt,
                    )
                    bot_reply = response.text
                    
                    # نمایش پاسخ در صفحه
                    st.markdown(bot_reply)
                    
                    # ذخیره پاسخ در حافظه موقت چت
                    st.session_state.messages.append({"role": "assistant", "content": bot_reply})
                
                except Exception as e:
                    st.error(f"خطایی در ارتباط با هوش مصنوعی رخ داد: {e}")
